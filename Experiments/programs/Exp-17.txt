#include <stdio.h>
#include <string.h>

char leading[3][20];

void add(char set[], char ch)
{
    int i;
    for(i = 0; set[i] != '\0'; i++)
    {
        if(set[i] == ch)
            return;
    }

    int len = strlen(set);
    set[len] = ch;
    set[len + 1] = '\0';
}

int main()
{
    strcpy(leading[0], "");
    strcpy(leading[1], "");
    strcpy(leading[2], "");

    // FIRST terminals for F
    add(leading[2], '(');
    add(leading[2], 'i');

    // LEADING(T) = LEADING(F)
    strcpy(leading[1], leading[2]);

    // LEADING(E) = LEADING(T)
    strcpy(leading[0], leading[1]);

    printf("Grammar:\n");
    printf("E -> E+T | T\n");
    printf("T -> T*F | F\n");
    printf("F -> (E) | i\n\n");

    printf("LEADING(E) = { ");
    for(int i = 0; leading[0][i] != '\0'; i++)
        printf("%c ", leading[0][i]);
    printf("}\n");

    printf("LEADING(T) = { ");
    for(int i = 0; leading[1][i] != '\0'; i++)
        printf("%c ", leading[1][i]);
    printf("}\n");

    printf("LEADING(F) = { ");
    for(int i = 0; leading[2][i] != '\0'; i++)
        printf("%c ", leading[2][i]);
    printf("}\n");

    return 0;
}
