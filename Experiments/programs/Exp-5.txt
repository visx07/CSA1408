#include <stdio.h>

int main()
{
    char ch;
    int spaces = 0, newlines = 0;

    printf("Enter text (Press Ctrl+D to stop):\n");

    while ((ch = getchar()) != EOF)
    {
        if (ch == ' ')
            spaces++;

        if (ch == '\n')
            newlines++;
    }

    printf("White Spaces = %d\n", spaces);
    printf("New Lines = %d\n", newlines);

    return 0;
}
