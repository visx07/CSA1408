#include <stdio.h>
#include <string.h>

struct Symbol
{
    char name[20];
    char type[20];
};

int main()
{
    struct Symbol table[20];
    int n, i;

    printf("Enter the number of symbols: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        printf("\nEnter Symbol %d Name: ", i + 1);
        scanf("%s", table[i].name);

        printf("Enter Symbol %d Type: ", i + 1);
        scanf("%s", table[i].type);
    }

    printf("\n------ SYMBOL TABLE ------\n");
    printf("Name\tType\n");

    for(i = 0; i < n; i++)
    {
        printf("%s\t%s\n", table[i].name, table[i].type);
    }

    return 0;
}
