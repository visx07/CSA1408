#include <stdio.h>
#include <string.h>

char trailing[3][20];

void add(char set[], char ch)
{
    int i;

    for(i = 0; set[i] != '\0'; i++)
    {
        if(set[i] == ch)
            return;
    }

    int len = strlen(set);
    set[len] = ch;
    set[len + 1] = '\0';
}

int main()
{
    strcpy(trailing[0], "");
    strcpy(trailing[1], "");
    strcpy(trailing[2], "");

    /* TRAILING(F) */
    add(trailing[2], ')');
    add(trailing[2], 'i');

    /* TRAILING(T) = TRAILING(F) */
    strcpy(trailing[1], trailing[2]);

    /* TRAILING(E) = TRAILING(T) */
    strcpy(trailing[0], trailing[1]);

    printf("Grammar:\n");
    printf("E -> E+T | T\n");
    printf("T -> T*F | F\n");
    printf("F -> (E) | i\n\n");

    printf("TRAILING(E) = { ");
    for(int i = 0; trailing[0][i] != '\0'; i++)
        printf("%c ", trailing[0][i]);
    printf("}\n");

    printf("TRAILING(T) = { ");
    for(int i = 0; trailing[1][i] != '\0'; i++)
        printf("%c ", trailing[1][i]);
    printf("}\n");

    printf("TRAILING(F) = { ");
    for(int i = 0; trailing[2][i] != '\0'; i++)
        printf("%c ", trailing[2][i]);
    printf("}\n");

    return 0;
}
