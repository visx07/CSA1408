#include <stdio.h>
#include <string.h>
#include <ctype.h>

char expr[100];
char stack[100];
int top = -1;
int temp = 1;

int precedence(char op)
{
    if (op == '*' || op == '/')
        return 2;

    if (op == '+' || op == '-')
        return 1;

    return 0;
}

void generateTAC(char op, char a, char b)
{
    printf("t%d = %c %c %c\n", temp, a, op, b);
}

int main()
{
    int i;
    char postfix[100];
    int p = 0;

    printf("Enter expression: ");
    scanf("%s", expr);

    /* Convert infix to postfix */
    for (i = 0; expr[i] != '\0'; i++)
    {
        if (isalnum(expr[i]))
        {
            postfix[p++] = expr[i];
        }
        else if (expr[i] == '(')
        {
            stack[++top] = expr[i];
        }
        else if (expr[i] == ')')
        {
            while (top != -1 && stack[top] != '(')
                postfix[p++] = stack[top--];

            top--;
        }
        else
        {
            while (top != -1 &&
                   precedence(stack[top]) >= precedence(expr[i]))
            {
                postfix[p++] = stack[top--];
            }

            stack[++top] = expr[i];
        }
    }

    while (top != -1)
        postfix[p++] = stack[top--];

    postfix[p] = '\0';

    printf("\nThree Address Code:\n");

    char operands[100][20];
    int opTop = -1;

    for (i = 0; postfix[i] != '\0'; i++)
    {
        if (isalnum(postfix[i]))
        {
            operands[++opTop][0] = postfix[i];
            operands[opTop][1] = '\0';
        }
        else
        {
            char b[20], a[20], result[20];

            strcpy(b, operands[opTop--]);
            strcpy(a, operands[opTop--]);

            sprintf(result, "t%d", temp);

            printf("%s = %s %c %s\n",
                   result, a, postfix[i], b);

            strcpy(operands[++opTop], result);

            temp++;
        }
    }

    printf("%s = %s\n", "Result", operands[opTop]);

    return 0;
}
